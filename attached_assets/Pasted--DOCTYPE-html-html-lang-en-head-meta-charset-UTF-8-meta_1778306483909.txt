// --- CORE DATA & NAVIGATION ---
let db = JSON.parse(localStorage.getItem('zapphub_db')) || [];

let activeClass = "";    
let selectedStudentClass = null; 

const adminCreds = { user: "Samuel Chibuike Azubuike", pass: "Lordmayor" };

function showScreen(id) {
  document.querySelectorAll('.screen').forEach(s => s.classList.add('hidden'));
  document.getElementById(id).classList.remove('hidden');
  if(id === 'student-portal') {
      selectedStudentClass = null; 
      renderStudentPortal();
  }
}

// --- CHAT APP LOGIC ---
let chatState = {
    contacts: [
        { id: 1, name: "Sarah", img: "https://i.pravatar.cc/150?u=1" },
        { id: 2, name: "John Doe", img: "https://i.pravatar.cc/150?u=2" },
        { id: 3, name: "Toolifylab Team", img: "https://i.pravatar.cc/150?u=3" }
    ],
    activeChats: []
};

function showChatView(viewId) {
    document.querySelectorAll('.view-chat-app').forEach(v => v.classList.remove('active'));
    document.getElementById(viewId).classList.add('active');
    if(viewId === 'view-chat-contacts') renderContacts();
    if(viewId === 'view-chat-dashboard') renderChatDashboard();
}

function renderContacts() {
    const list = document.getElementById('contacts-list');
    list.innerHTML = '';
    chatState.contacts.forEach(c => {
        const row = document.createElement('div');
        row.className = 'item-row';
        row.innerHTML = `<img src="${c.img}" class="avatar-sm"><div style="margin-left:15px"><h4>${c.name}</h4></div>`;
        row.onclick = () => startChat(c);
        list.appendChild(row);
    });
}

function startChat(contact) {
    if(!chatState.activeChats.find(c => c.id === contact.id)) chatState.activeChats.push(contact);
    document.getElementById('active-chat-name').innerText = contact.name;
    document.getElementById('active-chat-img').src = contact.img;
    document.getElementById('chat-messages').innerHTML = '';
    showChatView('view-chat-room');
}

function sendChatMessage() {
    const input = document.getElementById('chat-msg-input');
    if(!input.value.trim()) return;
    const msg = document.createElement('div');
    msg.className = 'bubble sent';
    msg.innerText = input.value;
    document.getElementById('chat-messages').appendChild(msg);
    input.value = '';
}

function renderChatDashboard() {
    const list = document.getElementById('chat-list');
    list.innerHTML = '';
    if(chatState.activeChats.length === 0) {
        list.innerHTML = '<p style="text-align:center; color:#888; margin-top:50px;">No active chats.</p>';
        return;
    }
    chatState.activeChats.forEach(c => {
        const row = document.createElement('div');
        row.className = 'item-row';
        row.innerHTML = `<img src="${c.img}" class="avatar-sm"><div style="margin-left:15px"><h4>${c.name}</h4><p style="font-size:12px; color:#666;">Tap to chat</p></div>`;
        row.onclick = () => startChat(c);
        list.appendChild(row);
    });
}

// --- ORIGINAL PORTAL & ADMIN LOGIC ---
function triggerUpload() { document.getElementById('profile-upload').click(); }
function handleProfilePic(input) {
  if (input.files && input.files[0]) {
    const reader = new FileReader();
    reader.onload = e => {
      document.querySelector('.profile-circle').style.backgroundImage = `url(${e.target.result})`;
      localStorage.setItem('zapphub_user_image', e.target.result);
    };
    reader.readAsDataURL(input.files[0]);
  }
}

function toggleTheme() {
    const theme = document.getElementById('dark-mode-toggle').checked ? 'dark' : 'light';
    document.body.setAttribute('data-theme', theme);
    localStorage.setItem('zapphub_theme', theme);
}

function showToast(m) {
  const t = document.createElement('div');
  t.className = 'toast'; t.innerText = m;
  document.getElementById('toast-container').appendChild(t);
  setTimeout(() => { t.style.opacity = '0'; setTimeout(() => t.remove(), 500); }, 3000);
}

function handleLogin() {
  const email = document.getElementById('login-email').value;
  if(email) document.getElementById('display-email').innerText = email;
  showScreen('user-dashboard');
}

function openAdminGate() { showScreen('admin-gate'); }
function verifyAdmin() {
  const u = document.getElementById('admin-user').value;
  const p = document.getElementById('admin-pass').value;
  if (u === adminCreds.user && p === adminCreds.pass) showScreen('admin-dashboard');
  else showToast("Unauthorized Access");
}

function showSection(type) {
  activeSection = type;
  document.querySelector('.main-portal-btns').classList.add('hidden');
  document.getElementById('class-selector').classList.remove('hidden');
  document.getElementById('current-section-title').innerText = type + " SECTIONS";
  const grid = document.getElementById('class-buttons-grid');
  grid.innerHTML = "";
  let classes = (type === 'Primary') ? ['Primary 1','Primary 2','Primary 3','Primary 4','Primary 5','Primary 6'] : 
                (type === 'JSS') ? ['JSS 1','JSS 2','JSS 3'] : ['SSS 1','SSS 2','SSS 3'];
  classes.forEach(c => {
    const btn = document.createElement('button'); btn.innerText = c;
    btn.onclick = () => selectClass(c); grid.appendChild(btn);
  });
}

function selectClass(c) {
  activeClass = c;
  document.getElementById('class-selector').classList.add('hidden');
  document.getElementById('term-module').classList.remove('hidden');
  document.getElementById('current-class-title').innerText = "Lessons for " + c;
  refreshUploadLists();
}

function uploadLesson(n) {
  const t = document.getElementById('title-'+n).value;
  const f = document.getElementById('file-'+n).files[0];
  if(!t || !f) return showToast("Fill all fields");
  db.push({ id: Date.now(), section: activeSection, class: activeClass, term: n, title: t, fileName: f.name });
  localStorage.setItem('zapphub_db', JSON.stringify(db));
  document.getElementById('title-'+n).value = ""; refreshUploadLists();
}

function refreshUploadLists() {
  for(let i=1; i<=3; i++) {
    const list = document.getElementById('list-' + i); list.innerHTML = "";
    db.filter(item => item.class === activeClass && item.term === i).forEach(item => {
      const div = document.createElement('div');
      div.style.cssText = "padding:10px; background:var(--term-content-bg); border:1px solid var(--reading-pane-border); margin-top:5px; border-radius:5px; display:flex; justify-content:space-between; align-items:center;";
      div.innerHTML = `<span>${item.title}</span> <button onclick="deleteItem(${item.id})" style="width:auto; padding:5px; background:red; color:white; font-size:10px;">Delete</button>`;
      list.appendChild(div);
    });
  }
}

function deleteItem(id) {
  db = db.filter(i => i.id !== id); localStorage.setItem('zapphub_db', JSON.stringify(db)); refreshUploadLists();
}

function adminBackward() {
  if (!document.getElementById('term-module').classList.contains('hidden')) backToClasses();
  else if (!document.getElementById('class-selector').classList.contains('hidden')) resetToMain();
  else showScreen('portal-selection');
}

function resetToMain() {
  document.getElementById('class-selector').classList.add('hidden');
  document.querySelector('.main-portal-btns').classList.remove('hidden');
}

function backToClasses() {
  document.getElementById('term-module').classList.add('hidden');
  document.getElementById('class-selector').classList.remove('hidden');
}

function toggleTerm(n) {
  const c = document.getElementById('content-'+n);
  const h = c.classList.contains('hidden');
  document.querySelectorAll('.term-content').forEach(x => x.classList.add('hidden'));
  if(h) c.classList.remove('hidden');
}

function renderStudentPortal(filter = "") {
  const container = document.getElementById('student-display');
  container.innerHTML = "";
  const filtered = db.filter(i => i.title.toLowerCase().includes(filter.toLowerCase()) || i.class.toLowerCase().includes(filter.toLowerCase()));
  if(!selectedStudentClass && filter === "") {
    const categories = [...new Set(db.map(i => i.class))];
    const grid = document.createElement('div'); grid.className = 'category-grid';
    categories.forEach(c => {
      const card = document.createElement('div'); card.className = 'category-card';
      card.innerHTML = `<span>${c}</span><span style="font-size:12px; color:gray">${db.filter(i=>i.class===c).length} Lessons</span>`;
      card.onclick = () => { selectedStudentClass = c; renderStudentPortal(); };
      grid.appendChild(card);
    });
    container.appendChild(grid);
  } else {
    const list = selectedStudentClass ? filtered.filter(i => i.class === selectedStudentClass) : filtered;
    list.forEach(i => {
      const card = document.createElement('div');
      card.style.cssText = "background:var(--card-bg); padding:15px; margin-bottom:10px; border-radius:10px; border-left:5px solid #3498db; text-align:left;";
      card.innerHTML = `<strong>${i.class} - T${i.term}</strong><br><span>${i.title}</span><br><button class="btn-read" onclick="openLesson(${i.id})">Read</button>`;
      container.appendChild(card);
    });
  }
}

function openLesson(id) {
  const l = db.find(i => i.id === id);
  if(!l) return;
  document.getElementById('view-title').innerText = l.title;
  document.getElementById('view-filename').innerText = l.fileName;
  showScreen('lesson-viewer');
}
